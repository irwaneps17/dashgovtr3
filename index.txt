    <script>
        const API_URL = "https://script.google.com/macros/s/AKfycbxVPzAxmQxlwOH6SAJV17BhOh6Inqz4R7m-479CoYgdhnnD051_1MmWXGGtqa8ojNTMg/exec";

        let barChartInstance = null;
        let doughnutChartInstance = null;

        async function fetchGsheetData() {
            try {
                const response = await fetch(API_URL);
                const rawData = await response.json();
                processAndRenderDashboard(rawData);
            } catch (error) {
                console.error("Gagal mengambil data dari Google Sheets:", error);
            }
        }

        function processAndRenderDashboard(data) {
            let totalProject = data.length;
            let totalNilaiBC = 0;
            let totalNilaiOBL = 0;
            let totalClose = 0;
            let totalInProgress = 0;

            let witelDataMap = {};

            data.forEach(item => {
                let witel = (item["Witel"] || "Lainnya").toString().trim();
                let nilaiBC = parseFloat(item["Nilai BC"]) || 0;
                let nilaiOBL = parseFloat(item["NILAI OBL (exc PPN)"]) || 0;
                let status = (item["Status Solution"] || "").toString().toLowerCase();

                totalNilaiBC += nilaiBC;
                totalNilaiOBL += nilaiOBL;

                if (status.includes("close") || status.includes("approve")) {
                    totalClose++;
                } else {
                    totalInProgress++;
                }

                // Grouping per Witel untuk Grafik Bar
                if (!witelDataMap[witel]) {
                    witelDataMap[witel] = 0;
                }
                witelDataMap[witel] += nilaiBC;
            });

            // Update Angka di Ringkasan Cards
            document.getElementById("valTotalProject").innerText = totalProject;
            document.getElementById("valNilaiBC").innerText = totalNilaiBC.toLocaleString("id-ID");
            document.getElementById("valNilaiOBL").innerText = totalNilaiOBL.toLocaleString("id-ID");
            document.getElementById("valProjectClose").innerText = totalClose;

            // Render Charts dengan Chart.js
            renderCharts(witelDataMap, totalClose, totalInProgress);
        }

        function renderCharts(witelMap, closeCount, inProgressCount) {
            // 1. Bar Chart Witel
            const witelLabels = Object.keys(witelMap);
            const witelValues = Object.values(witelMap);

            const ctxBar = document.getElementById('witelBarChart').getContext('2d');
            if (barChartInstance) barChartInstance.destroy();

            barChartInstance = new Chart(ctxBar, {
                type: 'bar',
                data: {
                    labels: witelLabels,
                    datasets: [{
                        label: 'Nilai BC (Rp)',
                        data: witelValues,
                        backgroundColor: '#3b82f6',
                        borderRadius: 6
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: { legend: { display: false } }
                }
            });

            // 2. Doughnut Chart Status
            const ctxDoughnut = document.getElementById('progressDoughnutChart').getContext('2d');
            if (doughnutChartInstance) doughnutChartInstance.destroy();

            doughnutChartInstance = new Chart(ctxDoughnut, {
                type: 'doughnut',
                data: {
                    labels: ['Closed / Approved', 'In Progress'],
                    datasets: [{
                        data: [closeCount, inProgressCount],
                        backgroundColor: ['#10b981', '#f59e0b'],
                        borderWidth: 0
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: {
                        legend: { position: 'bottom', labels: { boxWidth: 12, font: { size: 11 } } }
                    }
                }
            });
        }

        // Jalankan fungsi saat halaman selesai dimuat
        document.addEventListener("DOMContentLoaded", fetchGsheetData);
